\newFunc{bigO}{\ensuremath{\mathcal{O}}}{O(g(x))}{g(x)}{%

In other words, \bigThetab{g(x)}~describes an exact order of function growth%

\newFunc{geoMean}{\ensuremath{\operatorname{geom}}}{geom(A)}{A}{%
The \emph{geometric mean}~\geoMeanb{A} is the \scale\textsuperscript{th} root of the product of \scale\ \emph{positive} values in a dataset~$A=(\arrayIndex{a}{0},\arrayIndex{a}{1},\dots,\arrayIndex{a}{\scale-1})$ with $\arrayIndex{a}{i}>0$ for all~$i\in\intRange{0}{\scale}$, i.e., $\geoMeanb{A} = \sqrt[\scale]{\prod_{i=0}^{\scale-1} \arrayIndex{a}{i}}=\exp{\left(\frac{1}{\scale} \sum_{i=0}^{\scale-1} \ln{\arrayIndex{a}{i}} \right)}$. %
The formula using $\exp$ and $\ln$ is equivalent to the product formula. %
However, due to the limited range of floating point numbers on computers~\cite{IEEE2019ISFFPA}, the product formula may lead to an overflow or underflow and the $\exp$\nobreakdashes-$\ln$ method is more reliable. %
In~\cite{FW1986HNTLWSTCWTSBR}, it is shown that the geometric mean is the only reasonable measure of average when dealing with normalized data, where the \glslink{arithMean}{arithmetic mean} or \glslink{median}{median} become unreliable. %
In \python, you can use function \pythonIdx{statistics!geometric\_mean}\pythonilIdx{geometric_mean} from the module \pythonilIdx{statistics} to compute the geometric mean%
}%

\newFunc{littleO}{\ensuremath{\mathbf{o}}}{o(g(x))}{g(x)}{%
\mbox{If $f(x)=\littleOb{g(x)}$,} then there exist positive numbers~$x_0\in\realNumbersP$ such that~\mbox{$0\leq f(x)\leq c*g(x)$} for all~$x\geq x_0$ \emph{all}~$c\in\realNumbersP$~\cite{CLRS2009ITA:AN}. %
Different from \bigOb{g(x)}, \littleOb{g(x)} holds for \emph{all} constants~$c>0$, whereas~\bigOb{g(x)} implies the existence of \emph{some} constant~$c>0$. %
This basically means that $f(x)$ becomes insignificant compared to~$g(x)$ as~$x_0$ grows toward infinity. %
It is often written that~$\lim_{x_0\to\infty} \frac{f(x)}{g(x)} = 0$. %
In other words, \littleOb{g(x)}~describes an upper bound for function growth that is not asymptotically tight, i.e., an upper bound that grows faster than~$f(x)$%
}%
%
\newFunc{littleOmega}{\ensuremath{\mathbf{\omega}}}{w(g(x))}{g(x)}{%
\mbox{If $f(x)=\littleOmegab{g(x)}$,} then for every positive constant~$c>0$, there exists a number~$x_0\in\realNumbersP$ such that~\mbox{$0\leq c*g(x)< f(x)$} for all~$x\geq x_0$~\cite{CLRS2009ITA:AN}. %
In other words, $f(x)=\littleOmegab{g(x)}$ if and only if $g(x)=\littleOb{f(x)}$. %
In other words, \littleOmega\ describes a lower bound that is not asymptotically tight%
}%
%

\newFunc{median}{\ensuremath{\operatorname{median}}}{median(A)}{A}{%
The \emph{median}~\medianb{A} is the value separating the bigger-valued half from the smaller-valued half of a data sample or distribution. %
Its estimate is the value right in the middle of a \emph{sorted} data sample~$A=(\arrayIndex{a}{0},\arrayIndex{a}{1}, \dots, \arrayIndex{a}{\scale-1})$ where $\arrayIndex{a}{i-1}\leq \arrayIndex{a}{i} \; \forall i \in 1\dots (\scale-1)$ with an odd number of elements and the \arithMean\ of the two values in the middle if \scale\ is even. %
In other words, $\medianb{A} = \arrayIndex{a}{\frac{\scale-1}{2}}$ if \scale\ is odd and $\frac{1}{2}\left(\mathNoTopSpacing{\arrayIndex{a}{\frac{\scale}{2}-1} + \arrayIndex{a}{\frac{\scale}{2}}}\right)$ otherwise, i.e., if \scale\ is even%
}%

\newFunc{sampleVar}{\ensuremath{\operatorname{var}}}{var(A)}{A}{%
The \emph{variance} of a distribution is the expectation of the squared deviation of the underlying random variable from its mean. %
The variance~\sampleVarb{A} of a data sample~$A=(\arrayIndex{a}{0},\arrayIndex{a}{1}, \dots, \arrayIndex{a}{\scale-1})$ with \scale~observations can be estimated as~$\sampleVarb{A} = \frac{1}{\scale-1} \sum_{i=0}^{\scale-1} \left(\arrayIndex{a}{i} - \arithMeanb{A}\right)^2$. %
It can also be computed using~$\sampleVarb{A} = \left(\sum_{i=0}^{\scale-1} \arrayIndex{a}{i}^2\right)-\frac{1}{\scale}\left(\sum_{i=0}^{\scale-1} \arrayIndex{a}{i}\right)^2$, which affords computing it in a single pass, but this method may lead to disastrous rounding errors~\cite{CGL1983AFCTSVAAR}. %
A better choice is \citeauthor{W1962NOAMFCCSOSAP}'s algorithm~\cite{W1962NOAMFCCSOSAP,K1997SA}: %
It begins by setting its running mean~$m\gets0$, running variance~$v\gets0$, and its running counter~$i\gets0$. %
It then iterates over~$A$ with the current value being~\arrayIndex{a}{i}. %
In each step, it sets~$delta\gets \arrayIndex{a}{i}-m$, then updates~$m\gets m+\delta/(i+1)$, $v\gets v+\delta*(\arrayIndex{a}{i}-m)$, and finally~$i\gets i+1$. %
Once all the elements of~$A$ have been processed, the sample variance is~$\frac{1}{\scale-1}v$%
}%
